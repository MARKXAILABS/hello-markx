```js
const path = require('node:path'), fs = require('node:fs'), Module = require('node:module');
const loadTs = require('./load-ts.cjs');
// `@/*` -> `src/renderer/src/*` (tsconfig.web.json paths + electron.vite.config.ts:82, verified).
// loadTs() takes an exact path, so try the same candidates resolveTs() does.
const resolveAlias = (req) => {
  const base = path.join('src/renderer/src', req.slice(2));
  for (const c of [`${base}.tsx`, `${base}.ts`, `${base}/index.tsx`, `${base}/index.ts`]) {
    if (fs.existsSync(path.join(__dirname, '..', c))) return c;
  }
  return null;
};
const origLoad = Module._load;
Module._load = function (request, ...rest) {
  if (request.startsWith('@/')) {
    const hit = resolveAlias(request);
    if (hit) return loadTs(hit);              // real module, not a Proxy
  }
  return origLoad.call(this, request, ...rest);   // react, react-dom, everything else: untouched
};
```
